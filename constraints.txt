# constraints.txt for sptxinsight
#
# Provides version pins that prevent pip from scanning many candidate versions
# during dependency resolution (slow metadata fetches).
#
# HOW TO UPDATE: after a successful install run:
#   pip freeze | grep -E "boto3|aiobotocore|botocore|google-|zarr|anndata|scanpy|numpy|fsspec"
# and replace the ranges below with the exact resolved ==X.Y.Z pins.
#
# Ranges here are "good enough" to bound the search space; exact pins are
# faster but require a known-working install to capture.
# ─────────────────────────────────────────────────────────────────────────────

fastmcp>=3.4.5
fastmcp-slim>=3.4.5
# Core numpy<2 stack — must be kept in sync with pyproject.toml bounds
numpy<2.0.0
anndata>=0.11,<0.13   # 0.11 works with zarr 2.x; 0.13+ requires zarr>=3 which requires numpy>=2
zarr<3                 # zarr 3.x requires numpy>=2

# boto3 / aiobotocore — pip otherwise scans 90+ versions of boto3
# Replace with exact ==X.Y.Z after running pip freeze
# Floor is 1.40 (NOT 1.41): the shared wsinsight env (the master version set)
# locks boto3==1.40.49; a higher floor would force an upgrade in that env.
boto3>=1.40,<1.42
aiobotocore>=2.5.4,<3.0.0

# fsspec family
fsspec>=2023.1.0,<2026
s3fs<2026
gcsfs<2026

# scanpy — pin to last numpy<2-compatible release
scanpy<1.11

# --- PyTorch --------------------------------------------------------------
# pyproject leaves torch unpinned. Unconstrained it resolves to >=2.9, which
# pulls the CUDA 13 wheels; those overwrite nvidia/cudnn/lib and break the
# TensorFlow/StarDist half of the shared wsinsight env. torch 2.8.0 hard-pins
# its own nvidia-*-cu12 set, so no separate CUDA pins are needed here.
torch==2.8.0
torchvision==0.23.0
torch-geometric==2.6.1   # matches wsinsight's pin; used by niche discovery
